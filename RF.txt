/************************************************************
Conversion functions
************************************************************/
function dbToNatural(img) {
  return ee.Image(10).pow(img.divide(10));
}

function naturalToDb(img) {
  return ee.Image(img).log10().multiply(10);
}

/************************************************************
Speckle filtering
************************************************************/
function reduceSpeckleNoise(image) {

  var vvNat = dbToNatural(image.select('VV'));
  var vhNat = dbToNatural(image.select('VH'));

  var vvFiltered = vvNat.focal_median({
    radius: 20,
    units: 'meters'
  });

  var vhFiltered = vhNat.focal_median({
    radius: 20,
    units: 'meters'
  });

  return ee.Image.cat([

    naturalToDb(vvFiltered).rename('VV'),
    naturalToDb(vhFiltered).rename('VH')

  ]).copyProperties(image, image.propertyNames());
}

var s1 = ee.ImageCollection('COPERNICUS/S1_GRD')

  .filterBounds(studyArea)

  .filter(ee.Filter.eq('instrumentMode', 'IW'))

    'orbitProperties_pass',

  ))

  .select(['VV', 'VH']);

print('Total S1 images:', s1.size());

/************************************************************
Build Sentinel-1 composite
************************************************************/
function buildS1Composite(
  preStart,
  preEnd,
  postStart,
  postEnd,
  region
) {

  var preCol = s1.filterDate(preStart, preEnd);
  var postCol = s1.filterDate(postStart, postEnd);

  print(
    'S1 Pre image count (' + preStart + ' to ' + preEnd + '):',
    preCol.size()
  );

  print(
    'S1 Post image count (' + postStart + ' to ' + postEnd + '):',
    postCol.size()
  );

  print(
    'Pre dates:',
    preCol.aggregate_array('system:time_start').map(function(t) {
      return ee.Date(t).format('YYYY-MM-dd');
    })
  );

  print(
    'Post dates:',
    postCol.aggregate_array('system:time_start').map(function(t) {
      return ee.Date(t).format('YYYY-MM-dd');
    })
  );

  var pre = preCol
    .map(reduceSpeckleNoise)
    .median()
    .clip(region);

  var post = postCol
    .map(reduceSpeckleNoise)
    .median()
    .clip(region);

  return {
    pre: pre,
    post: post
  };
}

/************************************************************
Radar feature stack
************************************************************/
function buildS1Features(preImg, postImg) {

  var preVV = preImg
    .select('VV')
    .rename('preVV');

  var preVH = preImg
    .select('VH')
    .rename('preVH');

  var postVV = postImg
    .select('VV')
    .rename('postVV');

  var postVH = postImg
    .select('VH')
    .rename('postVH');

  var diffVV = preVV
    .subtract(postVV)
    .rename('diffVV');

  var diffVH = preVH
    .subtract(postVH)
    .rename('diffVH');

  

  return ee.Image.cat([

    preVV,
    preVH,

    postVV,
    postVH,

    diffVV,
    diffVH,



  ]);
}

  .fastDistanceTransform(1024)

  .multiply(ee.Image.pixelArea().sqrt())

/************************************************************

************************************************************/

/************************************************************
Area function
************************************************************/
function calcAreaKm2(maskImage, region) {

  var area = ee.Image.pixelArea()

    .updateMask(maskImage)

    .reduceRegion({

      reducer: ee.Reducer.sum(),

      geometry: region,

      scale: 10,

      maxPixels: 1e13

    })

    .values()

    .get(0);

  area = ee.Number(
    ee.Algorithms.If(area, area, 0)
  );

  return area.divide(1e6);
}

/************************************************************
Metrics
************************************************************/
var tpArea = calcAreaKm2(tp, compareRegion);
var fpArea = calcAreaKm2(fp, compareRegion);
var fnArea = calcAreaKm2(fn, compareRegion);
var tnArea = calcAreaKm2(tn, compareRegion);

var predictedArea = calcAreaKm2(
  predBin.selfMask(),
  compareRegion
);

var officialArea = calcAreaKm2(
  refBin.selfMask(),
  compareRegion
);

var precision = ee.Number(

  ee.Algorithms.If(

    tpArea.add(fpArea).gt(0),

    tpArea.divide(tpArea.add(fpArea)),

    0
  )
);

var recall = ee.Number(

  ee.Algorithms.If(

    tpArea.add(fnArea).gt(0),

    tpArea.divide(tpArea.add(fnArea)),

    0
  )
);

var f1 = ee.Number(

  ee.Algorithms.If(

    precision.add(recall).gt(0),

    precision
      .multiply(recall)
      .multiply(2)
      .divide(precision.add(recall)),

    0
  )
);

var iou = ee.Number(

  ee.Algorithms.If(

    tpArea.add(fpArea).add(fnArea).gt(0),

    tpArea.divide(
      tpArea.add(fpArea).add(fnArea)
    ),

    0
  )
);

var accuracy = ee.Number(

  ee.Algorithms.If(

    tpArea
      .add(fpArea)
      .add(fnArea)
      .add(tnArea)
      .gt(0),

    tpArea
      .add(tnArea)
      .divide(
        tpArea
          .add(fpArea)
          .add(fnArea)
          .add(tnArea)
      ),

    0
  )
);
