/************************************************************
 Conversion functions
 ************************************************************/
function dbToNatural(img) {
  return ee.Image(10).pow(img.divide(10));
}

function naturalToDb(img) {
  return ee.Image(img).log10().multiply(10);
}

/************************************************************
 Speckle filtering function
 ************************************************************/
function reduceSpeckleNoise(image) {
  var nat = dbToNatural(image);
  
  var filtered = nat.focal_median({
    radius: 20,
    units: 'meters'
  });
  
  return naturalToDb(filtered)
    .rename([pol])
    .copyProperties(image, image.propertyNames());
}

/************************************************************
 Otsu function
 ************************************************************/
var otsu = function(histogram) {
  histogram = ee.Dictionary(histogram);

  var counts = ee.Array(histogram.get('histogram'));
  var means = ee.Array(histogram.get('bucketMeans'));
  var size = means.length().get([0]);

  var total = counts.reduce(ee.Reducer.sum(), [0]).get([0]);
  var sum = means.multiply(counts).reduce(ee.Reducer.sum(), [0]).get([0]);
  var mean = sum.divide(total);

  var indices = ee.List.sequence(1, size);

  var bss = indices.map(function(i) {
    i = ee.Number(i);

    var aCounts = counts.slice(0, 0, i);
    var aCount = aCounts.reduce(ee.Reducer.sum(), [0]).get([0]);
    var aMeans = means.slice(0, 0, i);

    var aMean = ee.Algorithms.If(
      ee.Number(aCount).gt(0),
      aMeans.multiply(aCounts)
        .reduce(ee.Reducer.sum(), [0])
        .get([0])
        .divide(aCount),
      0
    );

    var bCount = ee.Number(total).subtract(aCount);

    var bMean = ee.Algorithms.If(
      bCount.gt(0),
      ee.Number(sum)
        .subtract(ee.Number(aCount).multiply(aMean))
        .divide(bCount),
      0
    );

    return ee.Number(aCount)
      .multiply(ee.Number(aMean).subtract(mean).pow(2))
      .add(
        bCount.multiply(
          ee.Number(bMean).subtract(mean).pow(2)
        )
      );
  });

  return means.sort(bss).get([-1]);
};

/************************************************************
 Compute one Otsu threshold
 ************************************************************/
function getOtsuThreshold(image, region, scale, bandName) {
  var hist = image.reduceRegion({
    reducer: ee.Reducer.histogram({
      maxBuckets: otsuMaxBuckets,
      minBucketWidth: otsuMinBucketWidth
    }),
    geometry: region,
    scale: scale,
    maxPixels: 1e13,
    bestEffort: true
  }).get(bandName);

  return ee.Number(otsu(hist));
}

/************************************************************
 Area function
 ************************************************************/
function calcAreaKm2(maskImage, region) {
  var areaDict = ee.Image.pixelArea()
    .updateMask(maskImage)
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13
    });
  
  var area = areaDict.get('area');
  area = ee.Number(ee.Algorithms.If(area, area, 0));
  return area.divide(1e6);

}

/************************************************************
 Metrics function
 ************************************************************/
function calcMetrics(predMask, refMask, region) {
  var pred = predMask.unmask(0);
  var ref = refMask.unmask(0);

  var tp = ee.Image.pixelArea()
    .updateMask(pred.and(ref))
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13
    })
    .values()
    .get(0);

  var fp = ee.Image.pixelArea()
    .updateMask(pred.and(ref.not()))
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13
    })
    .values()
    .get(0);

  var fn = ee.Image.pixelArea()
    .updateMask(pred.not().and(ref))
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13
    })
    .values()
    .get(0);

tp = ee.Number(ee.Algorithms.If(tp, tp, 0));
fp = ee.Number(ee.Algorithms.If(fp, fp, 0));
fn = ee.Number(ee.Algorithms.If(fn, fn, 0));

  var precision = ee.Algorithms.If(
    tp.add(fp).gt(0),
    tp.divide(tp.add(fp)),
    0
  );

  var recall = ee.Algorithms.If(
    tp.add(fn).gt(0),
    tp.divide(tp.add(fn)),
    0
  );

  var iou = ee.Algorithms.If(
    tp.add(fp).add(fn).gt(0),
    tp.divide(tp.add(fp).add(fn)),
    0
  );

  precision = ee.Number(precision);
  recall = ee.Number(recall);
  iou = ee.Number(iou);

  var f1 = ee.Algorithms.If(
    precision.add(recall).gt(0),
    precision.multiply(recall).multiply(2)
      .divide(precision.add(recall)),
    0
  );

  return ee.Dictionary({
    TP_km2: tp.divide(1e6),
    FP_km2: fp.divide(1e6),
    FN_km2: fn.divide(1e6),
    Precision: precision,
    Recall: recall,
    IoU: iou,
    F1: ee.Number(f1)
  });
}
