/************************************************************
Conversion functions
 ************************************************************/
function dbToNatural(img) {
  return ee.Image(10).pow(img.divide(10));
}

function naturalToDb(img) {
  return ee.Image(img).log10().multiply(10);
}

/************************************************************
Speckle filtering function
Apply filtering to each image before compositing
 ************************************************************/
function reduceSpeckleNoise(image) {
  var nat = dbToNatural(image);
  
  var filtered = nat.focal_median({
    radius: 20,
    units: 'meters'
  });
  
  return naturalToDb(filtered)
    .rename([pol])
    .copyProperties(image, image.propertyNames());
}

/************************************************************
Function to calculate area
 ************************************************************/
function calcAreaKm2(maskImage, region) {
  var areaDict = ee.Image.pixelArea()
    .updateMask(maskImage)
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13
    });
  
  var area = areaDict.get('area');
  area = ee.Number(ee.Algorithms.If(area, area, 0));
  return area.divide(1e6);
}

/************************************************************
Confusion metrics
 ************************************************************/
function calcMetrics(predMask, refMask, region) {
  var pred = predMask.unmask(0);
  var ref = refMask.unmask(0);

  var tp = ee.Image.pixelArea().updateMask(pred.and(ref)).reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: region,
    scale: 10,
    maxPixels: 1e13
  }).values().get(0);

  var fp = ee.Image.pixelArea().updateMask(pred.and(ref.not())).reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: region,
    scale: 10,
    maxPixels: 1e13
  }).values().get(0);

  var fn = ee.Image.pixelArea().updateMask(pred.not().and(ref)).reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: region,
    scale: 10,
    maxPixels: 1e13
  }).values().get(0);

  tp = ee.Number(tp);
  fp = ee.Number(fp);
  fn = ee.Number(fn);

  var precision = ee.Algorithms.If(
    tp.add(fp).gt(0),
    tp.divide(tp.add(fp)),
    0
  );

  var recall = ee.Algorithms.If(
    tp.add(fn).gt(0),
    tp.divide(tp.add(fn)),
    0
  );

  var iou = ee.Algorithms.If(
    tp.add(fp).add(fn).gt(0),
    tp.divide(tp.add(fp).add(fn)),
    0
  );

  precision = ee.Number(precision);
  recall = ee.Number(recall);
  iou = ee.Number(iou);

  var f1 = ee.Algorithms.If(
    precision.add(recall).gt(0),
    precision.multiply(recall).multiply(2).divide(precision.add(recall)),
    0
  );

  return ee.Dictionary({
    TP_km2: tp.divide(1e6),
    FP_km2: fp.divide(1e6),
    FN_km2: fn.divide(1e6),
    Precision: precision,
    Recall: recall,
    IoU: iou,
    F1: ee.Number(f1)
  });
}

var endDate = ee.Date(postEnd);

var numberOfDays = endDate.difference(startDate, 'day');

var includedDates = ee.List.sequence(0, numberOfDays.subtract(1))
  .map(function(day) {
    return startDate
      .advance(ee.Number(day), 'day')
      .format('YYYY-MM-dd');
  });

print('All dates INCLUDED in post-flood period:', includedDates);
