/************************************************************
Conversion functions
************************************************************/
function dbToNatural(img) {
  return ee.Image(10).pow(img.divide(10));
}

function naturalToDb(img) {
  return ee.Image(img).log10().multiply(10);
}

/************************************************************
Speckle filtering
************************************************************/
function reduceSpeckleNoise(image) {

  var vvNat = dbToNatural(
    image.select('VV')
  );

  var vhNat = dbToNatural(
    image.select('VH')
  );

  var vvFiltered = vvNat.focal_median({
    radius: 20,
    units: 'meters'
  });

  var vhFiltered = vhNat.focal_median({
    radius: 20,
    units: 'meters'
  });

  return ee.Image.cat([
    naturalToDb(vvFiltered).rename('VV'),
    naturalToDb(vhFiltered).rename('VH')
  ])
  .copyProperties(
    image,
    image.propertyNames()
  );
}

/************************************************************
Build Sentinel-1 composite
************************************************************/
function buildS1Composite(
  preStart,
  preEnd,
  postStart,
  postEnd,
  region
) {

  var preCol = s1.filterDate(
    preStart,
    preEnd
  );

  var postCol = s1.filterDate(
    postStart,
    postEnd
  );

  print(
    'S1 Pre image count (' +
    preStart + ' to ' + preEnd + '):',
    preCol.size()
  );

  print(
    'S1 Post image count (' +
    postStart + ' to ' + postEnd + '):',
    postCol.size()
  );

  print(
    'Pre dates:',
    preCol
      .aggregate_array('system:time_start')
      .map(function(time) {
        return ee.Date(time)
          .format('YYYY-MM-dd');
      })
  );

  print(
    'Post dates:',
    postCol
      .aggregate_array('system:time_start')
      .map(function(time) {
        return ee.Date(time)
          .format('YYYY-MM-dd');
      })
  );

  var pre = preCol
    .map(reduceSpeckleNoise)
    .median()
    .clip(region);

  var post = postCol
    .map(reduceSpeckleNoise)
    .median()
    .clip(region);

  return {
    pre: pre,
    post: post
  };
}

/************************************************************
Sentinel-1 feature stack
************************************************************/
function buildS1Features(
  preImg,
  postImg
) {

  var preVV = preImg
    .select('VV')
    .rename('preVV');

  var preVH = preImg
    .select('VH')
    .rename('preVH');

  var postVV = postImg
    .select('VV')
    .rename('postVV');

  var postVH = postImg
    .select('VH')
    .rename('postVH');

  var diffVV = preVV
    .subtract(postVV)
    .rename('diffVV');

  var diffVH = preVH
    .subtract(postVH)
    .rename('diffVH');

  return ee.Image.cat([
    preVV,
    preVH,
    postVV,
    postVH,
    diffVV,
    diffVH,
  ]);
}

var intercept =
  -0.10661359847066525;

var coefficients = [
   0.02167562281279925,   // diffVH
  -0.20962134671951674,   // diffVV
  -0.0014087285918233275, // postVH
  -0.8894862257006326,    // postVV
   0.009905091736059753,  // preVH
  -1.058309049430006      // preVV
];

var means = [
   0.037420645890683335,  // diffVH
  -0.04571752052474333,   // diffVV
 -15.875507831303333,     // postVH
  -8.800660985959334,     // postVV
 -15.83808716089,         // preVH
  -8.846378495015001      // preVV
];

var stds = [
  1.5317594585850032,     // diffVH
  1.6415157682468566,     // diffVV
  3.175247435699057,      // postVH
  3.5060062577133113,     // postVV
  2.9003962810171746,     // preVH
  3.2718617212380874      // preVV
];
/************************************************************
Standardize 2025 features
************************************************************/
var meanImg = ee.Image
  .constant(means)
  .rename(bands);

var stdImg = ee.Image
  .constant(stds)
  .rename(bands);

var coefImg = ee.Image
  .constant(coefficients)
  .rename(bands);

var standardized = testFeatures
  .select(bands)
  .subtract(meanImg)
  .divide(stdImg);

/************************************************************
Logistic Regression equation
z = intercept + sum(coefficient × standardized feature)
************************************************************/
var z = standardized
  .multiply(coefImg)
  .reduce(
    ee.Reducer.sum()
  )
  .add(intercept);

/************************************************************
Probability
p = 1 / (1 + exp(-z))
************************************************************/
var prob2025 = ee.Image(1)
  .divide(
    ee.Image(1)
      .add(
        z.multiply(-1).exp()
      )
  )
  .rename('probability');

/************************************************************
Automatic classification
************************************************************/
var rawFlood2025 = prob2025
  .gte(0.5)
  .rename('pred_raw');

/************************************************************
Morphological cleaning

Closing:
1. Dilation = focal_max
2. Erosion = focal_min

Radius = 10 metres
************************************************************/
var finalFlood2025 = rawFlood2025

  .focal_max({
    radius: morphologyRadius,
    units: 'meters'
  })

  .focal_min({
    radius: morphologyRadius,
    units: 'meters'
  })

  .rename('predictedFlood')

  .selfMask();

/************************************************************
Area calculation function
************************************************************/
function calcAreaKm2(
  maskImage,
  region
) {

  var area = ee.Image
    .pixelArea()
    .updateMask(maskImage)
    .reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: region,
      scale: 10,
      maxPixels: 1e13,
      tileScale: 4
    })
    .values()
    .get(0);

  area = ee.Number(
    ee.Algorithms.If(
      area,
      area,
      0
    )
  );

  return area.divide(1e6);
}

/************************************************************
Accuracy metrics
************************************************************/
var precision = ee.Number(
  ee.Algorithms.If(
    tpArea.add(fpArea).gt(0),

    tpArea.divide(
      tpArea.add(fpArea)
    ),

    0
  )
);

var recall = ee.Number(
  ee.Algorithms.If(
    tpArea.add(fnArea).gt(0),

    tpArea.divide(
      tpArea.add(fnArea)
    ),

    0
  )
);

var f1 = ee.Number(
  ee.Algorithms.If(
    precision.add(recall).gt(0),

    precision
      .multiply(recall)
      .multiply(2)
      .divide(
        precision.add(recall)
      ),

    0
  )
);

var iou = ee.Number(
  ee.Algorithms.If(
    tpArea
      .add(fpArea)
      .add(fnArea)
      .gt(0),

    tpArea.divide(
      tpArea
        .add(fpArea)
        .add(fnArea)
    ),

    0
  )
);

var accuracy = ee.Number(
  ee.Algorithms.If(
    tpArea
      .add(fpArea)
      .add(fnArea)
      .add(tnArea)
      .gt(0),

    tpArea
      .add(tnArea)
      .divide(
        tpArea
          .add(fpArea)
          .add(fnArea)
          .add(tnArea)
      ),

    0
  )
);
